'use strict';
// QA frame client (E-413 stage fields). It prints what a scratch hub sends and asserts nothing.
// node "$Q/fc.cjs" ['<features JSON array>'] < frames.jsonl     (run in the scratch project directory)
// Connects to ./.cland/hub.json's Unix socket as the human UI (the secret is never printed) with those features
// and waits for snapshot.end when "snapshot" is among them. Then sends each stdin line in order and waits up to 20 s
// for its result; ids q1, q2, ... are added when missing. Pseudo-frames:
//   {"qa":"wait","ms":N}                          pause N ms
//   {"qa":"pages","id":"x","req":{…},"hold":N}    task.list pages x1, x2, … each with the previous page's next as
//                                                 cursor, until next is null (at most 50); with hold, pause N ms
//                                                 after the first page
// Output: every frame whole, one JSON line each, in arrival order, with {"type":"qa-send","frame":…} before each
// request and {"type":"qa-end"} last. QA_STREAM=1 prints frames as they arrive (needed for a backgrounded client);
// otherwise all at the end. QA_SETTLE (default 700) ms after the last answer; QA_HOLD_MS=N then keeps listening up to
// N ms, ending early once the file named by QA_STOP exists.
const fs = require('node:fs');
const net = require('node:net');
const features = process.argv[2] ? JSON.parse(process.argv[2]) : [];
const input = fs.readFileSync(0, 'utf8').split('\n').filter(line => line.trim()).map(line => JSON.parse(line));
const discovery = JSON.parse(fs.readFileSync('.cland/hub.json', 'utf8'));
const env = process.env;
const frames = [];
let waiters = [];
let next = 0;
let done = false;
const sleep = ms => new Promise(resolve => setTimeout(resolve, ms));
function emit(frame) {
  if (done) return;
  frames.push(frame);
  if (env.QA_STREAM) process.stdout.write(`${JSON.stringify(frame)}\n`);
  for (const w of [...waiters]) if (w.test(frame)) { waiters = waiters.filter(x => x !== w); w.resolve(frame); }
}
function waitFor(test, ms) {
  const seen = frames.find(test);
  if (seen) return Promise.resolve(seen);
  return new Promise((resolve, reject) => {
    const w = { test, resolve: frame => { clearTimeout(timer); resolve(frame); } };
    const timer = setTimeout(() => { waiters = waiters.filter(x => x !== w); reject(new Error('timeout')); }, ms);
    waiters.push(w);
  });
}
const socket = net.createConnection(discovery.socket);
let buffer = '';
socket.setEncoding('utf8');
socket.on('data', chunk => {
  buffer += chunk;
  for (let nl = buffer.indexOf('\n'); nl >= 0; nl = buffer.indexOf('\n')) {
    const line = buffer.slice(0, nl).trim();
    buffer = buffer.slice(nl + 1);
    if (line) emit(JSON.parse(line));
  }
});

async function request(frame) {
  const sent = frame.id === undefined ? { ...frame, id: `q${++next}` } : frame;

  socket.write(`${JSON.stringify(sent)}\n`);
  try { return await waitFor(f => (f.type === 'result' && f.id === sent.id) || f.type === 'qa-closed', 20000); }
  catch { emit({ type: 'qa-timeout', id: sent.id }); return null; }
}
(async () => {
  try {
    await new Promise((resolve, reject) => { socket.once('connect', resolve); socket.once('error', reject); });
    socket.write(`${JSON.stringify({ type: 'hello', protocol: 2, client: 'ui', secret: discovery.secret, features })}\n`);
    const hello = await waitFor(f => ['welcome', 'error', 'qa-closed'].includes(f.type), 20000);
    if (hello.type === 'welcome' && features.includes('snapshot')) await waitFor(f => ['snapshot.end', 'snapshot.refused', 'qa-closed'].includes(f.type), 60000);
    for (const step of hello.type === 'welcome' ? input : []) {
      if (step.qa === 'wait') await sleep(step.ms);
      else if (step.qa === 'pages') {
        let cursor = null;
        for (let page = 1; page <= 50; page++) {
          const reply = await request({ ...step.req, type: 'task.list', id: `${step.id}${page}`, ...(cursor ? { cursor } : {}) });
          if (page === 1 && step.hold) await sleep(step.hold);
          cursor = reply && reply.ok ? reply.value.next : null;
          if (!cursor) break;
        }
      } else await request(step);
    }
    await sleep(Number(env.QA_SETTLE ?? 700));
    const until = Date.now() + Number(env.QA_HOLD_MS ?? 0);
    while (Date.now() < until && !(env.QA_STOP && fs.existsSync(env.QA_STOP))) await sleep(200);
  } catch (error) {
    emit({ type: 'qa-error', message: error.message });
    process.exitCode = 1;
  }
  emit({ type: 'qa-end' });
  done = true;
  if (!env.QA_STREAM) process.stdout.write(frames.map(f => JSON.stringify(f)).join('\n') + '\n');
  socket.destroy();
  setTimeout(() => process.exit(), 500).unref();
})();

## Probes that mislead

- Process probes exclude their own shell: use `pgrep`'s bracket trick or exclude the probe's process tree.
- Never treat the live project's `hub status` as evidence: the scratch hubs are the ones whose `.cland/hub.json` sits under `$Q/p1` and `$Q/p2`. `fc.cjs` and `drv.mjs` read `.cland/hub.json` from the current directory; confirm `pwd` before each scratch row.
- Model numbers are shared by every client in an epoch, and `topic`, `inbox`, `event`, `presence` and `session.row` frames can interleave with the frames a check causes: identify frames by item id, topic id, key or text, never by position or by "the last" line.
- Byte counts are the JSON re-serialization of the parsed frame, which equals the hub's own serialization; a reply-budget figure adds 1 for the newline. Measure only the frame named; the domain cap is per frame.
- The scratch copy has no dev history or landing receipts: a stage's `merged` count there is not evidence either way. Judge `total` and `statuses` live and `merged` through the Neovim and hub tests.
- Closed pages are ordered by numeric item identity across kinds (one counter numbers releases, epics and tasks), not by rank; a cursor names the last identity returned.
- Since T-1089 a fully closed container is not in the snapshot: `row(id)` in `q.cjs` is `undefined` for it (E-482, R-577 and the closed epics of R-377 at writing), and an expression that reads its fields there throws. Its roll-up comes with its row on its parent's closed page (E413S1-12, E413S1-26). A closed row is in the snapshot only as an ancestor of open work.
- `nvim/test/startup-budget-probe.ts` (E-1006's manual copied-tracker probe) is a script, never a suite test: it takes the source tracker directory as its argument and asserts macOS, so E413S1-27 runs it directly with `node`, sealed, after `quiet`, not through `nt`. It only copies `$W/.cland/tasks` into its own scratch project. Since T-1088 and T-1089 it asserts the open items plus their ancestor chain, the 120 KiB margin and the 128 KiB budget; its byte figure is the same reply E413S1-11 measures through the frame client, on a different scratch hub.
- A `task.comment` frame that names `as: {name: "rev", …}` for a name no session ever had is the scratch human's own verdict (the hub takes the role from its records, never from the frame). That is intended in E413S1-06 and E413S1-21.
- The driver sessions are unmanaged spokes: their merged rows have `launched: false`, and their rows are removed (a null row) when they disconnect.
- `QA_STREAM=1` captures are complete only after `wait`; the `qa-end` line marks the end. Without `QA_STREAM` a client prints everything at the end.
- The local `first completed hub.status` in E413S1-04 is a macOS figure under whatever load the machine has; T-1081 C6 names Linux and is judged on the Coordinator's Linux cover (E413S1-16). Speed criteria use measured wall times, never projections.
