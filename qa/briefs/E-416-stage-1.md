import { createRequire } from 'node:module';

  console.log(`archive-pressed ${await page.getByRole('button', { name: 'A · Archive hidden', exact: true }).getAttribute('aria-pressed')}`);
  console.log(`items-record ${await page.evaluate(() => localStorage.getItem('cland.ui.items.v1'))}`);
  console.log(`connection ${await page.getByRole('status', { name: 'Hub connection' }).textContent()}`);
  console.log(`off-origin ${offOrigin.length}`);
  await context.close();
} finally { await browser.close(); }
